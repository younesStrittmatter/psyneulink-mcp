# Seed list for scripts/generate_tools.py.
#
# Four directives are supported:
#   import-walk: <module>           walk the package's .py files; pull every
#                                   psyneulink symbol they import.
#   symbol:      <qualname>         include this symbol regardless.
#   package:     <module>           include every public class/function in this
#                                   package. Wide; use sparingly.
#   method:      <Class.method>     include a single bound method on a class;
#                                   the generated tool resolves the bound
#                                   instance from a `composition` kwarg before
#                                   dispatching the call. Used for surfaces
#                                   that aren't constructors (e.g.,
#                                   `Composition.add_node`).
#
# Lines starting with `#` and blank lines are ignored.

# --------------------------------------------------------------------------- #
# Full top-level PNL surface                                                  #
# --------------------------------------------------------------------------- #
# `package: psyneulink` walks every public class + function exposed at the
# top level of `psyneulink` (filtered to classes/functions by the
# introspection layer). PsyNeuLink intentionally re-exports its core types
# (Composition, every Mechanism subclass, every Function subclass,
# Projection types, Ports, Schedulers, …) at the top level so most of
# what an agent realistically constructs is reachable this way. The
# generator skips C-extensions and dynamically-built objects that
# `inspect.getsource` can't handle, so this remains safe to leave wide.
package: psyneulink
